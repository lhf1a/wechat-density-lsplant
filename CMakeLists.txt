cmake_minimum_required(VERSION 3.28)
project(sq2_density_test LANGUAGES C CXX ASM)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(LSPLANT_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(DEX_BUILDER_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(DOBBY_BUILD_EXAMPLE OFF CACHE BOOL "" FORCE)
set(DOBBY_BUILD_TEST OFF CACHE BOOL "" FORCE)
set(Plugin.SymbolResolver OFF CACHE BOOL "" FORCE)
add_subdirectory(${DEPS}/lsplant/lsplant/src/main/jni lsplant)
add_subdirectory(${DEPS}/dobby dobby EXCLUDE_FROM_ALL)
file(GLOB XDL_SOURCES ${DEPS}/xdl/xdl/src/main/cpp/*.c)
add_library(xdl STATIC ${XDL_SOURCES})
target_include_directories(xdl PUBLIC ${DEPS}/xdl/xdl/src/main/cpp/include)
add_library(sq2_density_test SHARED module.cpp)
target_include_directories(sq2_density_test PRIVATE ${DEPS}/zygisk/module/jni)
target_link_libraries(sq2_density_test PRIVATE lsplant_static dobby_static xdl log dl)
target_link_options(sq2_density_test PRIVATE -Wl,-z,max-page-size=65536 -Wl,--exclude-libs,ALL)
